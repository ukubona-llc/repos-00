"""
Ukubona Digital Twin - Prototype v1
Simulation + Gradient Update Engine
"""

import numpy as np
import uuid
import datetime
import random


# -------------------------------------------------
# Global Loss Function (Prototype)
# -------------------------------------------------

class GlobalLoss:

    def __init__(self, weights=None):
        if weights is None:
            weights = {
                "cost_overrun": 0.30,
                "schedule_delay": 0.25,
                "capital_idle": 0.15,
                "risk": 0.10,
                "profit": 0.20
            }
        self.weights = weights

    def compute(self, state):
        """
        L = weighted sum of system stress minus profit term
        """
        cost = state["cost_overrun"]
        delay = state["schedule_delay"]
        idle = state["capital_idle"]
        risk = state["risk_index"]
        profit = state["profit"]

        loss = (
            self.weights["cost_overrun"] * cost +
            self.weights["schedule_delay"] * delay +
            self.weights["capital_idle"] * idle +
            self.weights["risk"] * risk -
            self.weights["profit"] * profit
        )

        return loss


# -------------------------------------------------
# Digital Twin Core
# -------------------------------------------------

class DigitalTwin:

    def __init__(self):
        self.state = self.initialize_state()
        self.loss_fn = GlobalLoss()
        self.history = []

    def initialize_state(self):
        return {
            "capital_deployed": 4_000_000,
            "capital_idle": 1_000_000,
            "construction_progress": 0.45,
            "schedule_delay": 5,
            "cost_overrun": 200_000,
            "sales_pipeline": 1_200_000,
            "risk_index": 0.30,
            "profit": 0
        }

    def generate_event(self):
        events = [
            "shipment_delay",
            "permit_delay",
            "cost_spike",
            "sales_surge",
            "labor_shortage"
        ]
        event = random.choice(events)
        magnitude = random.uniform(0.01, 0.2)
        return event, magnitude

    def apply_event(self, event, magnitude):
        if event == "shipment_delay":
            self.state["schedule_delay"] += magnitude * 10
        elif event == "permit_delay":
            self.state["schedule_delay"] += magnitude * 5
            self.state["risk_index"] += magnitude * 0.05
        elif event == "cost_spike":
            self.state["cost_overrun"] += magnitude * 100000
        elif event == "sales_surge":
            self.state["profit"] += magnitude * 50000
            self.state["capital_idle"] -= magnitude * 20000
        elif event == "labor_shortage":
            self.state["schedule_delay"] += magnitude * 3
            self.state["cost_overrun"] += magnitude * 30000

    def compute_gradient(self):
        base_loss = self.loss_fn.compute(self.state)
        gradients = {}
        epsilon = 1e-4

        for key in self.state.keys():
            original = self.state[key]
            self.state[key] = original + epsilon
            new_loss = self.loss_fn.compute(self.state)
            grad = (new_loss - base_loss) / epsilon
            gradients[key] = grad
            self.state[key] = original

        return gradients

    def update(self, learning_rate=0.01):
        gradients = self.compute_gradient()
        for key in self.state.keys():
            self.state[key] -= learning_rate * gradients[key]
        loss = self.loss_fn.compute(self.state)
        self.log_state(loss, gradients)
        return loss

    def log_state(self, loss, gradients):
        entry = {
            "id": str(uuid.uuid4()),
            "timestamp": str(datetime.datetime.utcnow()),
            "state": self.state.copy(),
            "loss": loss,
            "gradients": gradients
        }
        self.history.append(entry)

    def step(self):
        event, magnitude = self.generate_event()
        self.apply_event(event, magnitude)
        loss_before = self.loss_fn.compute(self.state)
        loss_after = self.update()
        return {
            "event": event,
            "magnitude": magnitude,
            "loss_before": loss_before,
            "loss_after": loss_after,
            "state": self.state
        }

import numpy as np

class Agent:
    def __init__(self, name, role_type, sitg, visibility_mask):
        self.name = name
        self.role_type = role_type  # 'analyst', 'partner', 'regulator'
        self.sitg = sitg            # gamma_i (Skin-in-the-Game)
        self.mask = visibility_mask # Which state variables they influence

    def compute_local_gradient(self, state, loss_fn):
        grad = {}
        epsilon = 1e-4
        base_loss = loss_fn.compute(state)

        for key in self.mask:
            original = state[key]
            state[key] = original + epsilon
            new_loss = loss_fn.compute(state)
            grad[key] = (new_loss - base_loss) / epsilon
            state[key] = original
        return grad

class Organization:
    def __init__(self, agents, loss_fn):
        self.agents = agents
        self.loss_fn = loss_fn
        self.state = {
            "capital_deployed": 4.0, "capital_idle": 1.0,
            "progress": 0.45, "delay": 5.0, "risk": 0.3
        }

    def global_step(self, learning_rate=0.01):
        global_update = {k: 0 for k in self.state.keys()}

        for agent in self.agents:
            local_grad = agent.compute_local_gradient(self.state, self.loss_fn)
            for key, val in local_grad.items():
                global_update[key] += agent.sitg * val

        for key in self.state.keys():
            self.state[key] -= learning_rate * global_update[key]

        return self.state, self.loss_fn.compute(self.state)
