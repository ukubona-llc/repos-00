from flask import Flask, jsonify
import numpy as np, io, base64
from matplotlib.backends.backend_agg import FigureCanvasAgg

@app.route('/simulate')
def simulate():
    alpha = 0.01; target = 0.0

if __name__ == '__main__':
    app.run(debug=True)
